Jalankan ini di Supabase SQL Editor: 
create table if not exists public.candidate_records (
  id text primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  full_name text not null,
  rank_applied text,
  department text,
  rank_category text,
  vessel_type text,
  gt_kw text,
  coc_grade text,
  test_date text,
  proctor text,
  role_module text,
  char_correct integer default 0,
  tech_correct integer default 0,
  prac_correct integer default 0,
  time_taken text,
  cert_suitability text,
  oral_notes text
);

alter table public.candidate_records enable row level security;

create policy "Allow public access"
on public.candidate_records
for all
to anon
using (true)
with check (true);

alter publication supabase_realtime add table public.candidate_records;
